cmake_minimum_required(VERSION 3.20)
project(tickforge LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(TICKFORGE_BUILD_TESTS "Build C++ tests" OFF)
option(TICKFORGE_BUILD_BENCH "Build benchmarks" OFF)
option(TICKFORGE_BUILD_PYTHON "Build the Python extension" OFF)
option(TICKFORGE_SANITIZE "Build with AddressSanitizer and UBSan" OFF)

add_library(tickforge_core STATIC src/market_sim.cpp)
target_include_directories(tickforge_core PUBLIC include)
set_target_properties(tickforge_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(tickforge_core PRIVATE -Wall -Wextra -Wpedantic)

if(TICKFORGE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(tickforge_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()

if(TICKFORGE_BUILD_TESTS)
  enable_testing()
  add_executable(tickforge_tests tests/cpp/main.cpp tests/cpp/test_order_book.cpp tests/cpp/test_market_sim.cpp)
  target_include_directories(tickforge_tests SYSTEM PRIVATE third_party)
  target_link_libraries(tickforge_tests PRIVATE tickforge_core)
  target_compile_options(tickforge_tests PRIVATE -Wall -Wextra)
  add_test(NAME tickforge_tests COMMAND tickforge_tests)
endif()

if(TICKFORGE_BUILD_BENCH)
  add_executable(bench_order_book bench/bench_order_book.cpp)
  target_link_libraries(bench_order_book PRIVATE tickforge_core)
  # -march=native is not accepted by every compiler (e.g. some Apple Clang
  # versions on Apple Silicon), so only use it where it works.
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag(-march=native TICKFORGE_HAS_MARCH_NATIVE)
  target_compile_options(bench_order_book PRIVATE -O3)
  if(TICKFORGE_HAS_MARCH_NATIVE)
    target_compile_options(bench_order_book PRIVATE -march=native)
  endif()
endif()

if(TICKFORGE_BUILD_PYTHON OR SKBUILD)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core MODULE src/bindings.cpp)
  target_link_libraries(_core PRIVATE tickforge_core)
  install(TARGETS _core DESTINATION tickforge)
endif()
